/**
     * Checks if my snake is close with other snakes, and if it is, we check that our length is bigger than them.
     * Returns true if there is no snake nearby, and if there is if our snakes length is bigger that the enemys length.
     * Returns false if there is a snake nearby and their length is >= than our snake.
     * @param {Object} gameState - Current game position containging all info about the snake and the board.
     * @returns {boolean} - Returns true if no threatning opponent snake is nearby or if we are longer then the opponent 
     * snake, otherwise return false if their is an opponent with the length >= than ours.
     * 
     * @typedef {Object} position - Defines the board position properties.
     * @property {number} x - The x- coordinate of the game board.
     * @property {number} x - The y- coordinate of the game board.
     * 
     * @typedef {Object} snake - Defines generic type of all snakes properties.
     * @property {position[]} body - Array position of snake's body segments(head at index 0).
     * @property {number} length - The full length of the snake.
     * 
     * @typedef {Object} board - Defines the snake properties.
     * @property {snake[]} snakes - The total array of all snkes on the board.
     * 
     * @typedef {Object} you - Defines our snake 'you' properties.
     * @property {position[]} body - Our snake body segments.
     * @property {number} lenght  - Our snake body length.
     * 
     * @typedef {Object} GameState - Defines game state.
     * @property {Board} board - Board current state. 
     * @property {you} you  - Our snake's current state.
     * 
*/


function head_to_head_mov(gameState){
    const myHead =gameState.you.body[0];
    const myLength = gameState.you.length;


    for (const snake of gameState.board.snakes) {
        // Skip comparing with ourselves
        if (snake.body[0].x === myHead.x && snake.body[0].y === myHead.y) {
            return;
        }
        const opponentHead = snake.body[0];
        const opponentLength = snake.length;
        // Check for manhattan distance=1
        const isAdjacent = Math.abs(opponentHead.x - myHead.x) + 
        Math.abs(opponentHead.y -  myHead.y) ===1;
        // If opponent is larger or equal to our snake return false
        if(isAdjacent){
            if(myLength <=opponentLength){
                return false;
            }
        }

    }
    return true;
}
export default head_to_head_mov;
